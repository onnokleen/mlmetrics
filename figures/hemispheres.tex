\documentclass[border=5mm]{standalone}
\usepackage{tikz}
\usepackage{amsmath,amssymb}
\usetikzlibrary{positioning,arrows.meta}

\begin{document}
% Left-to-right map: inputs -> shared representation -> parameter-specific heads -> outputs.
\begin{tikzpicture}[
  font=\footnotesize,
  inputblock/.style={draw, rounded corners, fill=gray!10, align=center,
    minimum width=1.8cm, minimum height=1.4cm, text width=1.0cm, inner sep=4mm},
  coreblock/.style={draw, rounded corners, fill=red!10, align=center,
    minimum width=3.2cm, minimum height=1.8cm, text width=2.4cm, inner sep=4mm},
  meanblock/.style={draw, rounded corners, fill=yellow!15, align=center,
    minimum width=4.4cm, minimum height=1.8cm, text width=3.6cm, inner sep=4mm},
  varblock/.style={draw, rounded corners, fill=blue!15, align=center,
    minimum width=4.4cm, minimum height=1.8cm, text width=3.6cm, inner sep=4mm},
  flow/.style={thick,-{Latex},shorten >=1.5mm,shorten <=1.5mm},
  outputflow/.style={thick,-{Latex},shorten >=1.5mm,shorten <=3mm}
]
\node[inputblock] (x) {$\mathbf{x}_t$};
\node[coreblock, right=1.5cm of x] (core1) {Common core\\$L_c$ shared layers};
\node[meanblock, above right=0.7cm and 1.5cm of core1] (mean)
  {Mean hemisphere\\$L_m$ layers + linear head};
\node[varblock, below right=0.7cm and 1.5cm of core1] (var)
  {Variance hemisphere\\$L_v$ layers + softplus head};
\node[right=1.8cm of mean] (mu) {$\mu_t=f_{\phi_\mu}(\mathbf{x}_t)$};
\node[right=1.8cm of var] (sig2) {$\sigma_t^2=g_{\phi_\sigma}(\mathbf{x}_t)>0$};
\draw[flow] (x) -- (core1);
\draw[flow] (core1) -- (mean);
\draw[flow] (core1) -- (var);
\draw[outputflow] (mean) -- (mu);
\draw[outputflow] (var) -- (sig2);
\end{tikzpicture}
\end{document}
